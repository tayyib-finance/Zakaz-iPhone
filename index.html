// Функция нормализации текста для точного поиска ключевых слов function normalizeText(text) { if (!text) return ''; return text.toLowerCase() .replace(/gb/g, '') // убираем GB, чтобы легче сравнивать память .replace(/макс/g, 'max') // приводим русский "Макс" к "max" .replace(/iph|iphone/g, '') // убираем мусорные префиксы .replace(/burgun|burg/g, 'burgundy') // стандартизируем цвет Burgundy .replace(/[^a-z0-9 ]/g, ' ') // заменяем лишние символы на пробелы .replace(/\s+/g, ' ') // убираем лишние пробелы .trim(); } // 1. Парсер Ахмеда (ГЛАВНЫЙ): возвращает массив объектов строго по порядку function parseAhmed(text) { const lines = text.split('\n'); const result = []; lines.forEach(line => { if (!line.trim()) return; // Извлекаем цену в конце строки (все цифры после дефиса или пробела) const priceMatch = line.match(/(\d+)\s*$/); if (!priceMatch) return; const price = parseInt(priceMatch[1], 10); // Название модели — это всё, что до цены и дефиса let name = line.replace(/-\s*\d+\s*$/, '').trim(); // Извлекаем объем памяти для точного сопоставления (например, 256, 512, 1tb) const storageMatch = name.match(/(256|512|1tb|1тб)/i); const storage = storageMatch ? storageMatch[1].toLowerCase() : ''; result.push({ originalName: name, // Сохраняем красивое имя от Ахмеда searchStr: normalizeText(name), // Строка для поиска у других storage: storage, price: price }); }); return result; } // 2. Универсальный парсер для остальных поставщиков (Абдул, Дубай, Австралия) function parseOtherSuppliers(text) { const lines = text.split('\n'); const items = []; let currentHeader = ''; // Для контекста (например, если модель указана сверху, как у Дубая) lines.forEach(line => { const cleanLine = line.trim(); if (!cleanLine) return; // Ищем цену в строке: либо после @, либо после дефиса, либо просто в конце строки let price = null; let lineWithoutPrice = cleanLine; const atMatch = cleanLine.match(/@\s*(\d+)/); const dashMatch = cleanLine.match(/-\s*(\d+)\s*$/); const endMatch = cleanLine.match(/\s+(\d+)\s*$/); if (atMatch) { price = parseInt(atMatch[1], 10); lineWithoutPrice = cleanLine.split('@')[0]; } else if (dashMatch) { price = parseInt(dashMatch[1], 10); lineWithoutPrice = cleanLine.replace(/-\s*\d+\s*$/, ''); } else if (endMatch && !cleanLine.match(/^[a-zA-Z\s]+$/)) { // Проверяем, чтобы строка не была просто текстом, и берем число в конце price = parseInt(endMatch[1], 10); lineWithoutPrice = cleanLine.replace(/\s+\d+\s*$/, ''); } if (price !== null) { // Объединяем текущий заголовок (если есть) и текст строки без цены const fullText = currentHeader + ' ' + lineWithoutPrice; items.push({ searchStr: normalizeText(fullText), price: price }); } else { // Если в строке нет цены, возможно это заголовок модели (как "Iph 18 Pro 256GB" или "18 pro max 512GB🇦🇺") if (cleanLine.toLowerCase().includes('18 pro') || cleanLine.toLowerCase().includes('max')) { currentHeader = cleanLine; } } }); return items; } // 3. ГЛАВНАЯ ФУНКЦИЯ СРАВНЕНИЯ (Вызывать по кнопке "Сравнить цены") function comparePrices(ahmedRaw, abdulRaw, dubaiRaw, australiaRaw) { // Шаг 1: Базой всегда является Ахмед const baseProducts = parseAhmed(ahmedRaw); // Шаг 2: Парсим остальных поставщиков в плоские списки const abdulItems = parseOtherSuppliers(abdulRaw); const dubaiItems = parseOtherSuppliers(dubaiRaw); const australiaItems = parseOtherSuppliers(australiaRaw); // Вспомогательная функция поиска совпадений function findPriceForProduct(baseProduct, supplierItems) { // Ищем элемент, у которого в строке есть ключевые слова из базы Ахмеда // Проверяем совпадение по памяти (обязательно) и по остальным словам const match = supplierItems.find(item => { // Проверка памяти if (baseProduct.storage && !item.searchStr.includes(baseProduct.storage)) { return false; } // Проверяем, насколько слова из Ахмеда совпадают с позицией поставщика const baseWords = baseProduct.searchStr.split(' '); // Исключаем пустые слова const requiredWords = baseWords.filter(w => w.length > 0 && w !== baseProduct.storage); // Если это Max модель, проверяем наличие слова max if (baseProduct.searchStr.includes('max') && !item.searchStr.includes('max')) return false; if (!baseProduct.searchStr.includes('max') && item.searchStr.includes('max')) return false; // Считаем сколько ключевых слов (например, цвет: "black", "silver") совпало return requiredWords.every(word => item.searchStr.includes(word)); }); return match ? match.price : '—'; } // Шаг 3: Сопоставляем данные строго по списку Ахмеда const finalTable = baseProducts.map(prod => { return { model: prod.originalName, ahmedPrice: prod.price + ' ₽', abdulPrice: findPriceForProduct(prod, abdulItems) ? (findPriceForProduct(prod, abdulItems) + ' ₽') : '—', dubaiPrice: findPriceForProduct(prod, dubaiItems) ? (findPriceForProduct(prod, dubaiItems) + ' AED') : '—', australiaPrice: findPriceForProduct(prod, australiaItems) ? (findPriceForProduct(prod, australiaItems) + ' AED') : '—' }; }); return finalTable; }
